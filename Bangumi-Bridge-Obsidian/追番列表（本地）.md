// --- 工具函数 ---

// 从笔记内容提取字段
function extractField(content, field) {
  if (!content) return null;
  const esc = field.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  const re = new RegExp("^\\|\\s*" + esc + "\\s*\\|\\s*([^|\\n]+?)\\s*\\|", "m");
  const m = content.match(re);
  if (m) {
    return m[1].trim().replace(/`/g, '');
  }
  return null;
}

// 提取已观看集数
function extractWatchedEpisodes(content) {
  const match = content.match(/\*\*已观看集数\*\*：\s*(\d+)/);
  return match ? parseInt(match[1]) : 0;
}

// 从Frontmatter提取字段
function extractFrontmatter(content) {
  const frontmatterMatch = content.match(/^---\n([\s\S]*?)\n---/);
  if (!frontmatterMatch) return null;
  
  const yamlContent = frontmatterMatch[1];
  const frontmatter = {};
  
  yamlContent.split('\n').forEach(line => {
    const match = line.match(/^([^:]+):\s*(.+)$/);
    if (match) {
      let value = match[2].trim();
      // 移除引号
      if ((value.startsWith('"') && value.endsWith('"')) || 
          (value.startsWith("'") && value.endsWith("'"))) {
        value = value.substring(1, value.length - 1);
      }
      frontmatter[match[1].trim()] = value;
    }
  });
  
  return frontmatter;
}

// 从路径提取季度信息
function extractSeasonFromPath(filePath) {
  const match = filePath.match(/\/(\d{4})\/(\d{2})月新番/);
  if (match) {
    const year = match[1];
    const month = match[2];
    return `${year}年${month}月`;
  }
  return "未知季度";
}

// 检查下载文件夹更新状态
async function checkDownloadUpdate(filePath) {
  try {
    const file = app.vault.getAbstractFileByPath(filePath);
    if (!file) return { hasUpdate: false, error: "文件不存在" };
    
    const content = await app.vault.read(file);
    const frontmatter = extractFrontmatter(content);
    if (!frontmatter) {
      return { hasUpdate: false, error: "无法解析Frontmatter" };
    }
    
    const downloadPath = frontmatter.下载路径;
    if (!downloadPath || downloadPath === "无") {
      return { 
        hasUpdate: false, 
        error: "无下载路径",
        downloadPath: null,
        watchedEpisodes: 0,
        fileCount: 0,
        newCount: 0
      };
    }
    
    const watchedEpisodes = extractWatchedEpisodes(content);
    
    // 检查文件夹
    const fs = require('fs');
    const path = require('path');
    
    const cleanPath = downloadPath.replace(/`/g, '').trim();
    
    if (!fs.existsSync(cleanPath)) {
      return { 
        hasUpdate: false, 
        error: "文件夹不存在",
        downloadPath: cleanPath,
        watchedEpisodes,
        fileCount: 0,
        newCount: 0
      };
    }
    
    // 统计视频文件
    const files = fs.readdirSync(cleanPath);
    const videoExtensions = ['.mp4', '.mkv', '.avi', '.mov', '.wmv', '.flv', '.webm', '.m4v'];
    
    const videoFiles = files.filter(file => {
      if (file.startsWith('.')) return false; // 跳过隐藏文件
      const ext = path.extname(file).toLowerCase();
      return videoExtensions.includes(ext);
    });
    
    const fileCount = videoFiles.length;
    
    // 新增：计算更新集数（与第一个脚本保持一致）
    const newCount = Math.max(0, fileCount - watchedEpisodes); // 确保不会出现负数
    const hasUpdate = newCount > 0;
    
    return {
      hasUpdate,
      error: null,
      downloadPath: cleanPath,
      watchedEpisodes,
      fileCount,
      newCount: newCount
    };
    
  } catch (error) {
    console.error(`检查更新失败 ${filePath}:`, error);
    return { 
      hasUpdate: false, 
      error: error.message,
      downloadPath: null,
      watchedEpisodes: 0,
      fileCount: 0,
      newCount: 0
    };
  }
}

// 创建更新状态单元格（修改为显示更新集数）
function createUpdatestatusCell(updateInfo) {
  if (updateInfo.error === "无下载路径") {
    return dv.el("span", "未设置", { 
      attr: { 
        class: "status-no-path",
        title: "未设置下载路径"
      }
    });
  }
  
  if (updateInfo.error) {
    return dv.el("span", "❌失败", { 
      attr: { 
        class: "status-unknown",
        title: updateInfo.error
      }
    });
  }
  
  // 修改：显示更新集数，与第一个脚本保持一致
  if (updateInfo.hasUpdate) {
    return dv.el("span", `🆕已更新${updateInfo.newCount}集`, { 
      attr: { 
        class: "status-new",
        title: `已看: ${updateInfo.watchedEpisodes}集, 实际文件: ${updateInfo.fileCount}个`
      }
    });
  } else {
    return dv.el("span", `✅ 已同步`, { 
      attr: { 
        class: "status-normal",
        title: `已看: ${updateInfo.watchedEpisodes}集, 实际文件: ${updateInfo.fileCount}个`
      }
    });
  }
}

// 创建笔记链接
function createNoteLink(filePath, chineseName, hasUpdate) {
  return dv.el("a", chineseName, {
    href: filePath,
    attr: {
      class: `note-link ${hasUpdate ? 'pinned' : ''}`,
      "data-href": filePath,
      "data-tooltip-position": "top",
      "aria-label": `打开 ${chineseName}`,
      onclick: `(() => {
        app.workspace.openLinkText("${filePath}", "", false);
        return false;
      })()`
    }
  });
}

// 创建打开文件夹按钮
function createFolderButton(updateInfo) {
  if (!updateInfo.downloadPath || updateInfo.error) {
    return dv.el("button", "—", {
      attr: {
        class: "folder-btn disabled",
        disabled: true,
        title: updateInfo.error || "无下载路径"
      }
    });
  }
  
  return dv.el("button", "📁 打开", {
    attr: {
      class: "folder-btn",
      onclick: `(() => {
        const { shell } = require('electron');
        shell.openPath("${updateInfo.downloadPath.replace(/\\/g, '\\\\')}");
      })()`,
      title: `打开文件夹: ${updateInfo.downloadPath}`
    }
  });
}

// 主程序
try {
  // 收集所有bangumi笔记
  const pages = dv.pages().where(p => 
    p.file && 
    p.file.path && 
    p.file.path.startsWith(ROOT_FOLDER) &&
    p.tags && 
    (p.tags.includes("bangumi") || p.tags === "bangumi")
  ).array();
  
  const watchingItems = [];
  
  // 处理每个笔记
  for (const page of pages) {
    try {
      const filePath = page.file.path;
      const content = await app.vault.read(app.vault.getAbstractFileByPath(filePath));
      
      // 检查观看状态
      const frontmatter = extractFrontmatter(content);
      if (!frontmatter || frontmatter.观看状态 !== "在看") {
        continue; // 跳过非"在看"状态的笔记
      }
      
      const chineseName = frontmatter.中文名 || page.file.name.replace('.md', '');
      const season = extractSeasonFromPath(filePath);
      
      // 检查更新状态
      const updateInfo = await checkDownloadUpdate(filePath);
      
      watchingItems.push({
        filePath,
        fileName: page.file.name,
        chineseName,
        season,
        updateInfo,
        hasUpdate: updateInfo.hasUpdate,
        fileCount: updateInfo.fileCount,
        watchedEpisodes: updateInfo.watchedEpisodes,
        newCount: updateInfo.newCount
      });
      
    } catch (error) {
      console.error(`处理笔记失败 ${page.file.path}:`, error);
    }
  }
  
  // 统计信息
  const totalCount = watchingItems.length;
  const updateCount = watchingItems.filter(item => item.hasUpdate).length;
  
  dv.paragraph(`**📊 统计：** ${totalCount} 部番剧中，${updateCount} 部有更新\n`);
  
  // 排序：有更新的置顶，然后按更新集数降序，然后按季度排序（与第一个脚本保持一致）
  watchingItems.sort((a, b) => {
    // 1. 有更新的排在前面
    if (a.hasUpdate && !b.hasUpdate) return -1;
    if (!a.hasUpdate && b.hasUpdate) return 1;
    
    // 2. 如果都有更新，按更新集数从大到小排序
    if (a.hasUpdate && b.hasUpdate) {
      if (a.newCount > b.newCount) return -1;
      if (a.newCount < b.newCount) return 1;
    }
    
    // 3. 然后按季度排序（新季度在前）
    const seasonA = a.season;
    const seasonB = b.season;
    if (seasonA !== seasonB) {
      return seasonB.localeCompare(seasonA);
    }
    
    // 4. 最后按中文名排序
    return a.chineseName.localeCompare(b.chineseName, "zh-CN");
  });
  
  // 生成表格
  const tableRows = watchingItems.map(item => {
    return [
      // 中文名（链接到笔记）
      createNoteLink(item.filePath, item.chineseName, item.hasUpdate),
      
      // 开播季度
      dv.span(item.season, { 
        attr: { 
          style: "font-size: 12px;"
        }
      }),
      
      // 更新状态（现在显示更新集数）
      createUpdatestatusCell(item.updateInfo),
      
      // 打开文件夹
      createFolderButton(item.updateInfo)
    ];
  });
  
  // 显示表格
  dv.table(
    ["中文名", "开播季度", "更新状态", "打开文件夹"],
    tableRows
  );
  
  } catch (error) {
  dv.paragraph(`❌ 生成追番列表失败：${error.message}`);
  console.error("DataView脚本执行失败：", error);
}

## 说明
  **🎯 使用说明：**
  1. **点击中文名**：打开对应的番剧笔记
  2. **点击"打开文件夹"**：打开下载的视频文件夹
  3. **🆕 已更新 X 集**：表示有 X 集新文件未观看
  4. **✅ 已同步**：已观看所有下载的文件
  
  **🔢 排序规则：**
  1. 有更新的番剧自动置顶
  2. 更新集数多的排在前面
  3. 无更新的按开播季度（新→旧）排序
  4. 最后按中文名称排序
