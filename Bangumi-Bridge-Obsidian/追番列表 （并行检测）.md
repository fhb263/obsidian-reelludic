// 从Frontmatter提取字段
function extractFrontmatter(content) {
    if (!content) return null;
    
    const frontmatterMatch = content.match(/^---\n([\s\S]*?)\n---/);
    if (!frontmatterMatch) return null;
    
    const yamlContent = frontmatterMatch[1];
    const frontmatter = {};
    
    yamlContent.split('\n').forEach(line => {
        const match = line.match(/^([^:]+):\s*(.+)$/);
        if (match) {
            let value = match[2].trim();
            if ((value.startsWith('"') && value.endsWith('"')) || 
                (value.startsWith("'") && value.endsWith("'"))) {
                value = value.substring(1, value.length - 1);
            }
            frontmatter[match[1].trim()] = value;
        }
    });
    
    return frontmatter;
}

// 提取已观看集数
function extractWatchedEpisodes(content) {
    if (!content) return 0;
    
    const patterns = [
        /\*\*已观看集数\*\*：\s*(\d+(\.\d+)?)/,
        /已观看集数[:：]\s*(\d+(\.\d+)?)/,
        /watched.*?[:：]\s*(\d+(\.\d+)?)/i,
        /看到.*?[:：]\s*(\d+(\.\d+)?)/,
        /进度.*?[:：]\s*(\d+(\.\d+)?)/
    ];
    
    for (const pattern of patterns) {
        const match = content.match(pattern);
        if (match && match[1]) {
            const num = parseFloat(match[1]);
            if (!isNaN(num)) {
                return num;
            }
        }
    }
    
    return 0;
}

// 从正文中提取观看网址
function extractWatchUrl(content) {
    if (!content) return null;
    
    const markdownPattern = /\*\*观看网址\*\*：\s*\[[^\]]+\]\(([^)]+)\)/;
    const markdownMatch = content.match(markdownPattern);
    
    if (markdownMatch && markdownMatch[1]) {
        return markdownMatch[1].trim();
    }
    
    const urlPatterns = [
        /\*\*观看网址\*\*：\s*([^\s\n]+)/,
        /观看网址[:：]\s*([^\s\n]+)/
    ];
    
    for (const pattern of urlPatterns) {
        const match = content.match(pattern);
        if (match && match[1]) {
            return match[1].trim();
        }
    }
    
    return null;
}

// 从路径提取季度信息
function extractSeasonFromPath(filePath) {
    if (!filePath) return "未知季度";
    
    const patterns = [
        /\/(\d{4})\/(\d{2})月新番/,
        /\/(\d{4})年(\d{1,2})月/,
        /(\d{4})[-\/](\d{1,2})/
    ];
    
    for (const pattern of patterns) {
        const match = filePath.match(pattern);
        if (match) {
            const year = match[1];
            const month = match[2].padStart(2, '0');
            return `${year}年${month}月`;
        }
    }
    return "未知季度";
}

// 从HTML中提取最大集数
function extractMaxEpisodeFromHTML(html) {
    if (!html) return 0;
    
    const episodePatterns = [
        /第\s*(\d+)\s*集/g,
        /第\s*(\d+)\s*话/g,
        /Episode\s*(\d+)/gi,
        /ep\.\s*(\d+)/gi
    ];
    
    let maxEpisode = 0;
    
    for (const pattern of episodePatterns) {
        const matches = [...html.matchAll(pattern)];
        for (const match of matches) {
            const num = parseInt(match[1]);
            if (!isNaN(num) && num > 0 && num > maxEpisode) {
                maxEpisode = num;
            }
        }
    }
    
    return maxEpisode;
}

// 网站更新检测函数
async function checkWebsiteUpdate(watchUrl, content) {
    try {
        const actualWatchUrl = extractWatchUrl(content) || watchUrl;
        
        if (!actualWatchUrl || actualWatchUrl === "无" || actualWatchUrl === "未设置" || actualWatchUrl.trim() === "") {
            return { 
                hasUpdate: false, 
                error: "未设置观看网址",
                latestEpisode: 0,
                watchedEpisodes: 0,
                updateCount: 0
            };
        }
        
        const watchedEpisodes = extractWatchedEpisodes(content);
        
        let htmlContent = "";
        let error = null;
        
        try {
            const response = await requestUrl({
                url: actualWatchUrl,
                method: 'GET',
                headers: {
                    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
                },
                timeout: 10000
            });
            htmlContent = response.text;
        } catch (fetchError) {
            error = `网络请求失败: ${fetchError.message}`;
        }
        
        if (error) {
            return { 
                hasUpdate: false, 
                error: error,
                latestEpisode: 0,
                watchedEpisodes,
                updateCount: 0
            };
        }
        
        const latestEpisode = extractMaxEpisodeFromHTML(htmlContent);
        
        if (latestEpisode === 0) {
            return { 
                hasUpdate: false, 
                error: "未找到剧集信息",
                latestEpisode: 0,
                watchedEpisodes,
                updateCount: 0
            };
        }
        
        const updateCount = Math.max(0, latestEpisode - watchedEpisodes);
        const hasUpdate = updateCount > 0;
        
        return {
            hasUpdate,
            error: null,
            latestEpisode,
            watchedEpisodes,
            updateCount,
            watchUrl: actualWatchUrl
        };
        
    } catch (error) {
        return { 
            hasUpdate: false, 
            error: `检测失败: ${error.message}`,
            latestEpisode: 0,
            watchedEpisodes: 0,
            updateCount: 0
        };
    }
}

// 创建更新状态单元格
function createUpdatestatusCell(updateInfo) {

            attr: {

    if (updateInfo.error) {
        return dv.el("span", "❌失败", { 
            attr: { 
                class: "status-failed",
                title: updateInfo.error
            }
        });
    }
    
    if (updateInfo.hasUpdate) {
        return dv.el("span", `🆕已更新${updateInfo.updateCount}集`, { 
            attr: { 
                class: "status-update",
                title: `已看: ${updateInfo.watchedEpisodes}集, 最新: ${updateInfo.latestEpisode}集`
            }
        });
    } else {
        return dv.el("span", `✅ 已同步`, { 
            attr: { 
                class: "status-synced",
                title: `已看: ${updateInfo.watchedEpisodes}集, 最新: ${updateInfo.latestEpisode}集`
            }
        });
    }
}

function createNoteLink(filePath, chineseName, hasUpdate) {
    return dv.el("a", chineseName, {

            class: `note-link ${hasUpdate ? 'has-update' : ''}`,

            onclick: `app.workspace.openLinkText("${filePath}", "", false); return false;`
        }
    });
}

            onclick: `window.open("${updateInfo.watchUrl}", "_blank"); return false;`,

        }
    });
}

    );
    
    const updateResults = await Promise.all(updatePromises);
    
    // 第三步：合并检测结果
    for (let i = 0; i < watchingItems.length; i++) {
        watchingItems[i].updateInfo = updateResults[i];
        watchingItems[i].hasUpdate = updateResults[i].hasUpdate;
    }

    loadingElement.remove();
    
    // 统计信息
    const totalCount = watchingItems.length;
    const updateCount = watchingItems.filter(item => item.hasUpdate).length;
    const hasUrlCount = watchingItems.filter(item => 
        item.updateInfo.error !== "未设置观看网址").length;
    const failedCount = watchingItems.filter(item => 
        item.updateInfo.error && item.updateInfo.error !== "未设置观看网址").length;

	    if (!a.hasUpdate && b.hasUpdate) return 1;

	        if (a.updateInfo.updateCount > b.updateInfo.updateCount) return -1;
	        if (a.updateInfo.updateCount < b.updateInfo.updateCount) return 1;

	    const seasonB = b.season;
	    if (seasonA !== seasonB) {
	        return seasonB.localeCompare(seasonA);
    }
